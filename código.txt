export default function App() {
  const mountRef = useRef(null);
  
  // Estados em React
  const [gameState, setGameState] = useState('select_p1'); // select_p1, select_p2, start, question, dice, event, end
  const [p1Char, setP1Char] = useState(null);
  const [p2Char, setP2Char] = useState(null);

  const [player1Pos, setPlayer1Pos] = useState(0);
  const [player2Pos, setPlayer2Pos] = useState(0);
  const [currentPlayer, setCurrentPlayer] = useState(1);
  const [currentQuestion, setCurrentQuestion] = useState(null);
  const [feedback, setFeedback] = useState(null);
  const [specialEvent, setSpecialEvent] = useState(null); 
  const [winner, setWinner] = useState(null);
  const [diceValue, setDiceValue] = useState(1);
  const [isRolling, setIsRolling] = useState(false);
  
  const [cameraView, setCameraView] = useState('thirdPerson'); 
  const [isQuestionMinimized, setIsQuestionMinimized] = useState(false);
  const [isPaused, setIsPaused] = useState(false); 

  // Refs de sincronização com o Three.js
  const targetPositions = useRef([0, 0]);
  const cameraTargetPlayer = useRef(1);
  const cameraViewRef = useRef('thirdPerson');
  const reactionRef = useRef({ player: null, type: null, time: 0 }); 
  const p1CharRef = useRef('ripchip');
  const p2CharRef = useRef('topeira');
  const isThreeReady = useRef(false);

  // Inicializa o Motor 3D (Apenas quando sai da seleção de personagens)
  useEffect(() => {
    if (gameState === 'select_p1' || gameState === 'select_p2') return;

    let isMounted = true;
    let scene, camera, renderer, animationId;
    let p1Mesh, p2Mesh, tileCoords = [];
    let currentLookTarget = null; 

    // Controlo de Câmara
    let currentSpherical = null;
    let targetSpherical = null;
    let prevCameraView = 'thirdPerson';
    let isDragging = false;
    let previousMousePosition = { x: 0, y: 0 };

    const initThree = () => {
      if (!isMounted || !mountRef.current || isThreeReady.current) return;
      const THREE = window.THREE;
      
      currentLookTarget = new THREE.Vector3(0, 0, 0);
      currentSpherical = new THREE.Spherical(14, Math.PI / 3, 0);
      targetSpherical = new THREE.Spherical(14, Math.PI / 3, 0);

      // Cenário Base
      scene = new THREE.Scene();
      scene.background = new THREE.Color(0x38bdf8); // Céu mais vibrante
      scene.fog = new THREE.Fog(0x38bdf8, 50, 180); 

      camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
      
      renderer = new THREE.WebGLRenderer({ antialias: true });
      renderer.setSize(window.innerWidth, window.innerHeight);
      renderer.shadowMap.enabled = true;
      renderer.shadowMap.type = THREE.PCFSoftShadowMap; // Sombras mais suaves
      mountRef.current.innerHTML = '';
      mountRef.current.appendChild(renderer.domElement);

      const hemiLight = new THREE.HemisphereLight(0xffffff, 0x444444, 0.7);
      hemiLight.position.set(0, 200, 0);
      scene.add(hemiLight);

      const dirLight = new THREE.DirectionalLight(0xffffff, 1.2);
      dirLight.position.set(60, 120, 50);
      dirLight.castShadow = true;
      dirLight.shadow.mapSize.width = 2048;
      dirLight.shadow.mapSize.height = 2048;
      dirLight.shadow.camera.top = 100; dirLight.shadow.camera.bottom = -100;
      dirLight.shadow.camera.left = -100; dirLight.shadow.camera.right = 100;
      scene.add(dirLight);

      // Chão (Relva com material Standard)
      const ground = new THREE.Mesh(
        new THREE.PlaneGeometry(350, 350), 
        new THREE.MeshStandardMaterial({ color: 0x4ade80, roughness: 1, metalness: 0 })
      );
      ground.rotation.x = -Math.PI / 2;
      ground.receiveShadow = true;
      scene.add(ground);

      // Lago
      const lake = new THREE.Mesh(
        new THREE.CircleGeometry(30, 64), 
        new THREE.MeshStandardMaterial({ color: 0x3b82f6, roughness: 0.1, metalness: 0.2 })
      );
      lake.rotation.x = -Math.PI / 2;
      lake.position.set(-15, 0.1, -45);
      scene.add(lake);

      // Trilha Sinuosa
      const curve = new THREE.CatmullRomCurve3([
        new THREE.Vector3(0, 0, 15), new THREE.Vector3(12, 0, 0), new THREE.Vector3(30, 0, -5),
        new THREE.Vector3(25, 0, -25), new THREE.Vector3(0, 0, -35), new THREE.Vector3(-25, 0, -15),
        new THREE.Vector3(-35, 0, 15), new THREE.Vector3(-15, 0, 35), new THREE.Vector3(15, 0, 40),
        new THREE.Vector3(45, 0, 20), new THREE.Vector3(50, 0, -20), new THREE.Vector3(30, 0, -55),
      ]);
      tileCoords = curve.getPoints(TRACK_LENGTH);

      const createTileTexture = (num, type) => {
        const canvas = document.createElement('canvas');
        canvas.width = 128; canvas.height = 128;
        const ctx = canvas.getContext('2d');
        ctx.fillStyle = type === 'reset' ? '#ef4444' : type === 'advance' ? '#4ade80' : type === 'back' ? '#fb923c' : '#d97706';
        ctx.fillRect(0, 0, 128, 128);
        ctx.fillStyle = '#ffffff';
        ctx.font = 'bold 55px sans-serif';
        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
        ctx.fillText(num, 64, 64);
        return new THREE.CanvasTexture(canvas);
      };

      const tileGeo = new THREE.CylinderGeometry(2.8, 2.8, 0.6, 32);
      for(let i = 0; i <= TRACK_LENGTH; i++) {
        const isSpecial = SPECIAL_TILES[i];
        const type = isSpecial ? isSpecial.type : 'normal';
        const label = i === 0 ? "INÍCIO" : i === TRACK_LENGTH ? "FIM" : i.toString();
        
        // Face superior com textura, resto liso
        const sideMat = new THREE.MeshStandardMaterial({ color: 0x854d0e, roughness: 0.9 });
        const topMat = new THREE.MeshStandardMaterial({ map: createTileTexture(label, type), roughness: 0.8 });
        
        const tile = new THREE.Mesh(tileGeo, [sideMat, topMat, sideMat]);
        tile.position.copy(tileCoords[i]);
        tile.position.y = 0.3;
        tile.receiveShadow = true;
        tile.castShadow = true;
        scene.add(tile);
      }

      // Elementos Decorativos Fofos (Árvores e Arbustos)
      const createCuteTree = () => {
        const tree = new THREE.Group();
        const trunk = new THREE.Mesh(new THREE.CylinderGeometry(0.6, 0.8, 3, 12), new THREE.MeshStandardMaterial({color: 0x78350f, roughness: 1}));
        trunk.position.y = 1.5; trunk.castShadow = true;
        
        const leavesMat = new THREE.MeshStandardMaterial({color: 0x166534, roughness: 0.9});
        const leaf1 = new THREE.Mesh(new THREE.SphereGeometry(2.5, 16, 16), leavesMat);
        leaf1.position.set(0, 4, 0); leaf1.castShadow = true;
        const leaf2 = new THREE.Mesh(new THREE.SphereGeometry(1.8, 16, 16), leavesMat);
        leaf2.position.set(1.5, 3.2, 0.5); leaf2.castShadow = true;
        const leaf3 = new THREE.Mesh(new THREE.SphereGeometry(1.8, 16, 16), leavesMat);
        leaf3.position.set(-1.2, 3.5, -1); leaf3.castShadow = true;
        
        tree.add(trunk, leaf1, leaf2, leaf3);
        return tree;
      };

      const createBush = () => {
        const bush = new THREE.Mesh(new THREE.SphereGeometry(1.5, 12, 12), new THREE.MeshStandardMaterial({color: 0x15803d, roughness: 0.9}));
        bush.scale.set(1, 0.7, 1);
        bush.position.y = 0.8;
        bush.castShadow = true;
        return bush;
      };

      for(let i=0; i<80; i++) {
        const x = (Math.random() - 0.5) * 200; 
        const z = (Math.random() - 0.5) * 200;
        if(!tileCoords.some(pt => pt.distanceTo(new THREE.Vector3(x, 0, z)) < 8)) {
          const isBush = Math.random() > 0.6;
          const prop = isBush ? createBush() : createCuteTree();
          prop.position.set(x, 0, z);
          const s = 0.5 + Math.random() * 0.6; 
          prop.scale.set(s,s,s);
          scene.add(prop);
        }
      }

      // --- GERADOR DE PERSONAGENS 3D (Estilo Peões Standee) ---
      // Em vez de formas geométricas feias, usamos um peão bonito com a ilustração virada para a câmara
      const buildCharacterStandee = (id) => {
        const charData = CHARACTERS[id] || CHARACTERS['ripchip'];
        const group = new THREE.Group();

        // 1. Base do Peão (Plástico/Madeira)
        const baseGeo = new THREE.CylinderGeometry(1.4, 1.6, 0.4, 32);
        const baseMat = new THREE.MeshStandardMaterial({ color: charData.color, roughness: 0.8, metalness: 0.2 });
        const base = new THREE.Mesh(baseGeo, baseMat);
        base.position.y = 0.2;
        base.castShadow = true;
        group.add(base);

        // 2. Anel Dourado Decorativo
        const ringGeo = new THREE.TorusGeometry(1.45, 0.1, 16, 32);
        const ringMat = new THREE.MeshStandardMaterial({ color: 0xfbbf24, metalness: 0.8, roughness: 0.2 });
        const ring = new THREE.Mesh(ringGeo, ringMat);
        ring.position.y = 0.4;
        ring.rotation.x = Math.PI / 2;
        group.add(ring);

        // 3. Sprite do Personagem (Sempre virado para a câmara, visual perfeito)
        const canvas = document.createElement('canvas');
        canvas.width = 256; canvas.height = 256;
        const ctx = canvas.getContext('2d');
        
        // Efeito de sombra no ícone
        ctx.shadowColor = 'rgba(0,0,0,0.5)';
        ctx.shadowBlur = 10;
        ctx.shadowOffsetY = 8;
        
        ctx.font = '160px sans-serif';
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillText(charData.icon, 128, 140); // Desenha o Emoji

        const texture = new THREE.CanvasTexture(canvas);
        const spriteMat = new THREE.SpriteMaterial({ map: texture, transparent: true });
        const sprite = new THREE.Sprite(spriteMat);
        sprite.scale.set(5, 5, 1);
        sprite.position.y = 2.8; // Flutua sobre a base
        group.add(sprite);

        return group;
      };

      p1Mesh = buildCharacterStandee(p1CharRef.current);
      p2Mesh = buildCharacterStandee(p2CharRef.current);
      
      p1Mesh.position.copy(tileCoords[0]); p1Mesh.position.x -= 1.2; 
      p2Mesh.position.copy(tileCoords[0]); p2Mesh.position.x += 1.2;
      scene.add(p1Mesh); scene.add(p2Mesh);

      // --- EVENTOS DE RATO (CONTROLO DE CÂMARA LIVRE) ---
      const onPointerDown = (e) => {
        isDragging = true;
        previousMousePosition = { x: e.clientX, y: e.clientY };
        if(cameraViewRef.current !== 'free') setCameraView('free');
      };
      const onPointerMove = (e) => {
        if (isDragging) {
          const deltaX = e.clientX - previousMousePosition.x;
          const deltaY = e.clientY - previousMousePosition.y;
          targetSpherical.theta -= deltaX * 0.01;
          targetSpherical.phi -= deltaY * 0.01;
          targetSpherical.phi = Math.max(0.1, Math.min(Math.PI / 2 - 0.1, targetSpherical.phi)); 
          previousMousePosition = { x: e.clientX, y: e.clientY };
        }
      };
      const onPointerUp = () => { isDragging = false; };
      const onWheel = (e) => {
        e.preventDefault();
        targetSpherical.radius += e.deltaY * 0.02;
        targetSpherical.radius = Math.max(5, Math.min(80, targetSpherical.radius));
        if(cameraViewRef.current !== 'free') setCameraView('free');
      };

      const canvas = renderer.domElement;
      canvas.style.touchAction = 'none'; // Previne scroll no mobile
      canvas.addEventListener('pointerdown', onPointerDown);
      window.addEventListener('pointermove', onPointerMove);
      window.addEventListener('pointerup', onPointerUp);
      canvas.addEventListener('wheel', onWheel, { passive: false });

      isThreeReady.current = true;

      // Loop de Animação
      const renderLoop = () => {
        animationId = requestAnimationFrame(renderLoop);

        const updatePlayerVisual = (mesh, targetIndex, offset, playerNum) => {
          const idx = Math.min(targetIndex, TRACK_LENGTH);
          const targetPos = tileCoords[idx].clone();
          targetPos.x += offset; 
          
          mesh.position.lerp(targetPos, 0.08); 
          
          const reaction = reactionRef.current;
          let isReacting = false;

          if (reaction.player === playerNum) {
            const elapsed = Date.now() - reaction.time;
            if (elapsed < 1500) {
              isReacting = true;
              const progress = elapsed / 1500;
              if (reaction.type === 'success') {
                // Salto e rotação felizes
                mesh.position.y = Math.abs(Math.sin(progress * Math.PI)) * 5; 
                mesh.rotation.y += 0.2; 
              } else if (reaction.type === 'fail') {
                // "Esmagamento" de erro
                const squash = 1 - Math.sin(progress * Math.PI) * 0.5;
                mesh.scale.set(1/squash, Math.max(0.2, squash), 1/squash);
                mesh.rotation.z = Math.sin(progress * Math.PI * 10) * 0.3;
              }
            }
          }

          if (!isReacting) {
            mesh.scale.lerp(new THREE.Vector3(1,1,1), 0.1); 
            mesh.rotation.z *= 0.8; 

            if (mesh.position.distanceTo(targetPos) > 0.5) {
              // Efeito de salto a caminhar
              mesh.position.y = Math.abs(Math.sin(Date.now() * 0.015)) * 2;
              // NOTA: Como usamos Sprites 2D, a rotação em Y afeta apenas a base, o boneco está sempre a olhar para nós! Fica perfeito.
              mesh.rotation.y = Math.atan2(targetPos.x - mesh.position.x, targetPos.z - mesh.position.z);
            } else {
              mesh.position.y = 0;
              if (idx < TRACK_LENGTH) {
                 const nextTile = tileCoords[idx + 1];
                 mesh.rotation.y = Math.atan2(nextTile.x - mesh.position.x, nextTile.z - mesh.position.z);
              }
            }
          }
        };

        updatePlayerVisual(p1Mesh, targetPositions.current[0], -1.2, 1);
        updatePlayerVisual(p2Mesh, targetPositions.current[1], 1.2, 2);

        // --- ATUALIZAÇÃO DA CÂMARA ---
        const activeIdx = cameraTargetPlayer.current === 1 ? targetPositions.current[0] : targetPositions.current[1];
        const activePos = tileCoords[Math.min(activeIdx, TRACK_LENGTH)];
        const nextIdx = Math.min(activeIdx + 1, TRACK_LENGTH);
        const dir = new THREE.Vector3().subVectors(tileCoords[nextIdx], activePos).normalize();
        if(dir.lengthSq() < 0.1) dir.set(0, 0, 1);

        if (prevCameraView !== cameraViewRef.current) {
          prevCameraView = cameraViewRef.current;
          switch(cameraViewRef.current) {
            case 'thirdPerson':
              targetSpherical.radius = 18; targetSpherical.phi = Math.PI / 3;
              targetSpherical.theta = Math.atan2(-dir.x, -dir.z);
              break;
            case 'secondPerson':
              targetSpherical.radius = 14; targetSpherical.phi = Math.PI / 2.5;
              targetSpherical.theta = Math.atan2(dir.x, dir.z);
              break;
            case 'topDown':
              targetSpherical.radius = 60; targetSpherical.phi = 0.05;
              break;
          }
        }

        let idealCamPos = new THREE.Vector3();
        let idealLookTarget = new THREE.Vector3();

        if (cameraViewRef.current === 'firstPerson') {
          idealCamPos.copy(activePos).add(new THREE.Vector3(0, 6, 0));
          idealLookTarget.copy(activePos).add(new THREE.Vector3(0, 6, 0)).add(dir.clone().multiplyScalar(20));
          camera.position.lerp(idealCamPos, 0.1);
        } else {
          currentSpherical.radius += (targetSpherical.radius - currentSpherical.radius) * 0.1;
          currentSpherical.phi += (targetSpherical.phi - currentSpherical.phi) * 0.1;
          
          let diff = targetSpherical.theta - currentSpherical.theta;
          while (diff < -Math.PI) diff += Math.PI * 2;
          while (diff > Math.PI) diff -= Math.PI * 2;
          currentSpherical.theta += diff * 0.1;

          const offset = new THREE.Vector3().setFromSpherical(currentSpherical);
          idealCamPos.copy(activePos).add(offset);
          idealLookTarget.copy(activePos).add(new THREE.Vector3(0, 2, 0));
          
          camera.position.copy(idealCamPos);
        }

        currentLookTarget.lerp(idealLookTarget, 0.1);
        camera.lookAt(currentLookTarget);

        renderer.render(scene, camera);
      };

      renderLoop();

      const handleResize = () => {
        camera.aspect = window.innerWidth / window.innerHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
      };
      window.addEventListener('resize', handleResize);

      return () => {
        window.removeEventListener('resize', handleResize);
        canvas.removeEventListener('pointerdown', onPointerDown);
        window.removeEventListener('pointermove', onPointerMove);
        window.removeEventListener('pointerup', onPointerUp);
        canvas.removeEventListener('wheel', onWheel);
        cancelAnimationFrame(animationId);
        renderer.dispose();
      };
    };

    if (!window.THREE) {
      const script = document.createElement('script');
      script.src = "https://cdnjs.cloudflare.com/ajax/libs/three.js/r128/three.min.js";
      script.onload = () => { if (isMounted) initThree(); };
      document.head.appendChild(script);
    } else {
      initThree();
    }

    return () => { isMounted = false; };
  }, [gameState === 'select_p1', gameState === 'select_p2']);

  // Sincronizar React State com Refs
  useEffect(() => { targetPositions.current[0] = player1Pos; targetPositions.current[1] = player2Pos; }, [player1Pos, player2Pos]);
  useEffect(() => { cameraTargetPlayer.current = currentPlayer; }, [currentPlayer]);
  useEffect(() => { cameraViewRef.current = cameraView; }, [cameraView]);
  useEffect(() => { p1CharRef.current = p1Char; p2CharRef.current = p2Char; }, [p1Char, p2Char]);

  const selectCharacter = (id) => {
    if (gameState === 'select_p1') {
      setP1Char(id);
      setGameState('select_p2');
    } else if (gameState === 'select_p2') {
      setP2Char(id);
      setGameState('start');
    }
  };

  const startGame = () => {
    setPlayer1Pos(0); setPlayer2Pos(0); setCurrentPlayer(1);
    setWinner(null); setFeedback(null); setSpecialEvent(null);
    setGameState('question');
    setIsQuestionMinimized(false);
    setIsPaused(false);
    pickRandomQuestion();
  };

  const pickRandomQuestion = () => {
    setCurrentQuestion(QUESTIONS[Math.floor(Math.random() * QUESTIONS.length)]);
  };

  const handleAnswer = (selectedAnswer) => {
    if (gameState !== 'question' || feedback || isPaused) return;
    const isCorrect = selectedAnswer === currentQuestion.answer;
    
    reactionRef.current = { player: currentPlayer, type: isCorrect ? 'success' : 'fail', time: Date.now() };

    if (isCorrect) {
      setFeedback({ isCorrect: true, msg: "Fantástico! Gira o dado!" });
      setTimeout(() => { setFeedback(null); setGameState('dice'); }, 1500);
    } else {
      setFeedback({ isCorrect: false, msg: `Ops! A resposta era ${formatAnswerName(currentQuestion.answer)}.` });
      setTimeout(() => nextTurn(), 2500);
    }
  };

  const rollDice = () => {
    if (isRolling || isPaused) return;
    setIsRolling(true);
    let rolls = 0;
    const rollInterval = setInterval(() => {
      setDiceValue(Math.floor(Math.random() * 6) + 1);
      rolls++;
      if (rolls >= 15) {
        clearInterval(rollInterval);
        setIsRolling(false);
        const finalValue = Math.floor(Math.random() * 6) + 1;
        setDiceValue(finalValue);
        handleDiceResult(finalValue);
      }
    }, 80);
  };

  const handleDiceResult = (rolledValue) => {
    setFeedback({ isCorrect: true, msg: `Tiraste ${rolledValue}! A avançar...` });
    setTimeout(() => {
      const currentPos = currentPlayer === 1 ? player1Pos : player2Pos;
      const nextPos = Math.min(currentPos + rolledValue, TRACK_LENGTH);
      updatePlayerPosition(currentPlayer, nextPos);

      if (nextPos >= TRACK_LENGTH) {
        setTimeout(() => finishGame(currentPlayer), 2000); 
        return;
      }
      setTimeout(() => {
        if (SPECIAL_TILES[nextPos]) triggerSpecialEvent(nextPos);
        else nextTurn();
      }, 1500); 
    }, 1500);
  };

  const triggerSpecialEvent = (landedPos) => {
    const event = SPECIAL_TILES[landedPos];
    setSpecialEvent(event);
    setGameState('event');
    setFeedback(null);
    
    setTimeout(() => {
      let finalPos;
      if (event.type === 'reset') finalPos = 0;
      else finalPos = Math.min(Math.max(landedPos + event.value, 0), TRACK_LENGTH);
      updatePlayerPosition(currentPlayer, finalPos);

      setTimeout(() => {
        if (finalPos >= TRACK_LENGTH) finishGame(currentPlayer);
        else nextTurn();
      }, 2000);
    }, 2500);
  };

  const updatePlayerPosition = (player, newPos) => {
    if (player === 1) setPlayer1Pos(newPos); else setPlayer2Pos(newPos);
  };

  const nextTurn = () => {
    setFeedback(null); setSpecialEvent(null);
    setIsQuestionMinimized(false);
    setCurrentPlayer(currentPlayer === 1 ? 2 : 1);
    setGameState('question');
    pickRandomQuestion();
  };

  const finishGame = (winningPlayer) => {
    setWinner(winningPlayer); setGameState('end'); setIsPaused(false);
  };

  const formatAnswerName = (key) => {
    const map = { contagem: "Contagem", ordenacao: "Ordenação", medida: "Medida", codigo: "Código" };
    return map[key] || key;
  };

  const DadoUI = ({ face, isRolling }) => {
    const dots = [];
    for(let i=0; i<face; i++) dots.push(<div key={i} className="w-3 h-3 bg-white rounded-full m-1 shadow-inner"/>);
    return (
      <div className={`w-20 h-20 bg-red-600 rounded-2xl flex flex-wrap items-center justify-center p-2 border-b-4 border-r-4 border-red-800 shadow-xl mx-auto my-4 transition-transform ${isRolling ? 'animate-[spin_0.2s_linear_infinite]' : 'scale-110'}`}>
         {dots}
      </div>
    );
  };

  // Renderização da Tela de Seleção
  if (gameState === 'select_p1' || gameState === 'select_p2') {
    const isP1 = gameState === 'select_p1';
    return (
      <div className="min-h-screen bg-sky-100 flex flex-col items-center justify-center p-4 font-sans">
        <div className="bg-white rounded-3xl shadow-2xl p-8 max-w-4xl w-full text-center border-8 border-green-600 relative overflow-hidden">
          {/* Decoração de fundo */}
          <div className="absolute top-0 left-0 w-full h-32 bg-green-500 opacity-20 transform -skew-y-3 origin-top-left"></div>
          
          <h1 className="text-4xl md:text-5xl font-black text-slate-800 mb-2 uppercase relative z-10">Aventura Nárnia 3D</h1>
          <h2 className={`text-xl md:text-2xl font-bold mb-8 px-8 py-3 inline-block rounded-full shadow-lg text-white transition-colors duration-500 relative z-10 ${isP1 ? 'bg-blue-500' : 'bg-red-500'}`}>
            Jogador {isP1 ? '1' : '2'}, escolhe o teu Personagem!
          </h2>
          
          <div className="grid grid-cols-2 md:grid-cols-5 gap-4 relative z-10">
            {Object.values(CHARACTERS).map((char) => {
              const isSelected = (!isP1 && p1Char === char.id);
              return (
                <button 
                  key={char.id} 
                  disabled={isSelected}
                  onClick={() => selectCharacter(char.id)}
                  className={`flex flex-col items-center justify-center p-4 rounded-2xl border-b-8 transition-all shadow-md
                    ${isSelected ? 'opacity-40 grayscale cursor-not-allowed bg-gray-100 border-gray-300' 
                                 : 'bg-white border-amber-400 hover:bg-amber-50 hover:-translate-y-2 hover:shadow-xl active:translate-y-0 active:border-b-0'}
                  `}
                >
                  <span className="text-6xl mb-3 filter drop-shadow-md">{char.icon}</span>
                  <span className="font-extrabold text-slate-700 text-sm md:text-base">{char.name}</span>
                </button>
              );
            })}
          </div>
        </div>
      </div>
    );
  }

  // Define se devemos mostrar o UI de jogo (escondido durante Start ou Fim)
  const isPlaying = ['question', 'dice', 'event'].includes(gameState);

  // --- RENDERIZAÇÃO PRINCIPAL DO JOGO 3D ---
  return (
    <div className="relative w-full h-screen overflow-hidden font-sans bg-slate-900 select-none">
      
      {/* Contentor do Mundo 3D */}
      <div ref={mountRef} className="absolute inset-0 z-0 pointer-events-auto" />

      {/* Interface Overlay */}
      <div className="absolute inset-0 z-10 pointer-events-none flex flex-col">
        
        {/* Ecrã de Pausa (Bloqueia Interações) */}
        {isPaused && (
          <div className="absolute inset-0 z-50 bg-black/60 backdrop-blur-sm flex items-center justify-center pointer-events-auto">
            <div className="bg-white p-10 rounded-3xl shadow-2xl text-center border-8 border-amber-500 max-w-sm w-full animate-in zoom-in">
              <h2 className="text-4xl font-black text-slate-800 mb-6 uppercase">Jogo Pausado</h2>
              <button onClick={() => setIsPaused(false)} className="bg-amber-500 hover:bg-amber-400 text-white px-8 py-4 rounded-full font-black text-xl flex items-center justify-center gap-3 w-full border-b-8 border-amber-700 active:border-b-0 active:translate-y-2 transition-all">
                <Play fill="currentColor" size={24} /> RETOMAR
              </button>
            </div>
          </div>
        )}

        {/* Barra de Controlo de Jogo (Fundo do ecrã) */}
        {gameState !== 'end' && (
          <div className="absolute bottom-6 left-1/2 transform -translate-x-1/2 z-40 flex gap-2 md:gap-4 pointer-events-auto bg-white/90 backdrop-blur-md px-4 md:px-6 py-3 rounded-full shadow-2xl border-2 border-slate-200">
            {gameState === 'start' ? (
              <button onClick={startGame} className="flex items-center gap-2 text-green-700 hover:text-green-500 font-bold transition-colors px-2">
                <Play size={20} fill="currentColor" /> <span className="hidden md:inline">Iniciar</span>
              </button>
            ) : (
              <button onClick={() => setIsPaused(!isPaused)} className="flex items-center gap-2 text-amber-700 hover:text-amber-500 font-bold transition-colors px-2">
                {isPaused ? <><Play size={20} fill="currentColor" /> <span className="hidden md:inline">Retomar</span></> : <><Pause size={20} fill="currentColor" /> <span className="hidden md:inline">Pausar</span></>}
              </button>
            )}
            
            <div className="w-px bg-slate-300 mx-1"></div>

            <button onClick={() => { startGame(); }} className="flex items-center gap-2 text-blue-700 hover:text-blue-500 font-bold transition-colors px-2">
              <RotateCcw size={20} /> <span className="hidden md:inline">Reiniciar</span>
            </button>
            
            <div className="w-px bg-slate-300 mx-1"></div>

            <button onClick={() => { setGameState('select_p1'); setIsPaused(false); }} className="flex items-center gap-2 text-red-700 hover:text-red-500 font-bold transition-colors px-2">
              <Square size={20} fill="currentColor" /> <span className="hidden md:inline">Terminar</span>
            </button>
          </div>
        )}

        {/* Elementos de UI Apenas Visíveis Durante a Partida Ativa */}
        {isPlaying && (
          <>
            {/* Painel de Controlo da Câmara */}
            <div className="absolute top-4 left-4 z-20 flex flex-col gap-2 pointer-events-auto">
              <button onClick={() => setCameraView('free')} className={`px-4 py-2 rounded-xl text-xs md:text-sm font-bold shadow-lg flex items-center gap-2 transition-all border ${cameraView === 'free' ? 'bg-amber-500 border-amber-600 text-white' : 'bg-white/90 border-slate-200 text-slate-700 hover:bg-white'}`}>
                <MousePointer2 size={16} /> Câmara Livre
              </button>
              <button onClick={() => setCameraView('thirdPerson')} className={`px-4 py-2 rounded-xl text-xs md:text-sm font-bold shadow-lg flex items-center gap-2 transition-all border ${cameraView === 'thirdPerson' ? 'bg-indigo-600 border-indigo-700 text-white' : 'bg-white/90 border-slate-200 text-slate-700 hover:bg-white'}`}>
                <User size={16} /> 3ª Pessoa (Atrás)
              </button>
              <button onClick={() => setCameraView('firstPerson')} className={`px-4 py-2 rounded-xl text-xs md:text-sm font-bold shadow-lg flex items-center gap-2 transition-all border ${cameraView === 'firstPerson' ? 'bg-indigo-600 border-indigo-700 text-white' : 'bg-white/90 border-slate-200 text-slate-700 hover:bg-white'}`}>
                <Eye size={16} /> 1ª Pessoa
              </button>
              <button onClick={() => setCameraView('secondPerson')} className={`px-4 py-2 rounded-xl text-xs md:text-sm font-bold shadow-lg flex items-center gap-2 transition-all border ${cameraView === 'secondPerson' ? 'bg-indigo-600 border-indigo-700 text-white' : 'bg-white/90 border-slate-200 text-slate-700 hover:bg-white'}`}>
                <Camera size={16} /> 2ª Pessoa (Frente)
              </button>
              <button onClick={() => setCameraView('topDown')} className={`px-4 py-2 rounded-xl text-xs md:text-sm font-bold shadow-lg flex items-center gap-2 transition-all border ${cameraView === 'topDown' ? 'bg-indigo-600 border-indigo-700 text-white' : 'bg-white/90 border-slate-200 text-slate-700 hover:bg-white'}`}>
                <MapIcon size={16} /> Visão Aérea
              </button>
            </div>

            {/* Status Superior Direito */}
            <div className="absolute top-4 right-4 z-20 flex flex-col items-end gap-3 pointer-events-auto">
              <div className="bg-white/90 backdrop-blur-md px-6 py-3 rounded-2xl shadow-lg border-2 border-white/50 text-right">
                <div className="flex flex-col gap-2 text-sm font-bold">
                  <span className={`px-3 py-1 rounded-md transition-colors ${currentPlayer === 1 ? 'bg-blue-500 text-white shadow-md scale-105' : 'text-slate-500 bg-slate-100'}`}>
                    {CHARACTERS[p1Char].icon} P1: Casa {player1Pos}
                  </span>
                  <span className={`px-3 py-1 rounded-md transition-colors ${currentPlayer === 2 ? 'bg-red-500 text-white shadow-md scale-105' : 'text-slate-500 bg-slate-100'}`}>
                    {CHARACTERS[p2Char].icon} P2: Casa {player2Pos}
                  </span>
                </div>
              </div>
              <div className={`px-6 py-2 rounded-full shadow-lg border-2 transition-colors duration-300
                ${currentPlayer === 1 ? 'bg-blue-600 border-blue-400 text-white' : 'bg-red-600 border-red-400 text-white'}`}>
                <span className="font-black uppercase tracking-widest text-xs md:text-sm">Vez do Jogador {currentPlayer}</span>
              </div>
            </div>
          </>
        )}

        {/* Painel de Perguntas LATERAL DIREITO */}
        {gameState === 'question' && (
          <div className="absolute right-4 top-1/2 transform -translate-y-1/2 w-80 md:w-96 flex flex-col items-end pointer-events-auto z-30">
            {!isQuestionMinimized ? (
              <div className="w-full bg-amber-100/95 backdrop-blur-md p-6 rounded-3xl shadow-2xl border-4 border-amber-600 animate-in slide-in-from-right relative">
                
                <button onClick={() => setIsQuestionMinimized(true)} className="absolute top-4 right-4 bg-amber-200 hover:bg-amber-300 p-2 rounded-full text-amber-800 transition-colors">
                  <EyeOff size={20} />
                </button>

                <h3 className="text-amber-900 font-bold text-center mb-4 uppercase tracking-widest text-xs opacity-80 border-b-2 border-amber-200 pb-2 mr-6">
                  Classifica o número:
                </h3>
                
                <p className="text-xl md:text-2xl font-black text-slate-800 text-center mb-6 bg-white p-4 rounded-2xl shadow-inner">
                  "{currentQuestion?.text}"
                </p>

                {!feedback ? (
                  <div className="flex flex-col gap-3">
                    {['contagem', 'ordenacao', 'medida', 'codigo'].map((opcao) => (
                      <button key={opcao} onClick={() => handleAnswer(opcao)} className="bg-white border-b-4 border-amber-300 hover:border-amber-400 text-amber-900 font-black py-3 rounded-xl shadow-md active:translate-y-1 transition-all text-sm uppercase">
                        {formatAnswerName(opcao)}
                      </button>
                    ))}
                  </div>
                ) : (
                  <div className={`p-4 rounded-2xl border-4 text-center ${feedback.isCorrect ? 'bg-green-100 border-green-500 text-green-800' : 'bg-red-100 border-red-500 text-red-800'}`}>
                    {feedback.isCorrect ? <CheckCircle size={32} className="mx-auto mb-2 text-green-600" /> : <XCircle size={32} className="mx-auto mb-2 text-red-600" />}
                    <p className="text-lg font-black">{feedback.msg}</p>
                  </div>
                )}
              </div>
            ) : (
               <button onClick={() => setIsQuestionMinimized(false)} className="bg-amber-500 hover:bg-amber-400 text-white p-4 rounded-2xl shadow-2xl border-b-4 border-amber-700 font-black flex flex-col items-center gap-2 animate-bounce">
                <Eye size={28} /> VER PERGUNTA
               </button>
            )}
          </div>
        )}

        {/* Área Central para Outros Eventos (Início, Dado, Vitória) */}
        <div className="flex-grow flex items-center justify-center p-4">
          
          {gameState === 'start' && (
            <div className="bg-white/95 backdrop-blur-xl p-8 rounded-[2rem] shadow-2xl border-4 border-green-500 max-w-xl text-center pointer-events-auto">
              <div className="flex justify-center items-center gap-6 text-6xl mb-6 font-black">
                <span className="filter drop-shadow-md">{CHARACTERS[p1Char].icon}</span>
                <span className="text-3xl text-slate-400">VS</span>
                <span className="filter drop-shadow-md">{CHARACTERS[p2Char].icon}</span>
              </div>
              <h2 className="text-3xl font-black text-slate-800 mb-4 uppercase">
                {CHARACTERS[p1Char].name} <br/>vs<br/> {CHARACTERS[p2Char].name}
              </h2>
              <p className="text-slate-600 text-lg font-medium mb-8 bg-slate-100 p-4 rounded-xl shadow-inner">
                🖱️ <b>Dica:</b> Clica e arrasta no cenário para rodar a câmara. Usa a roda do rato para dar Zoom! A pergunta ficará na lateral.
              </p>
              <button onClick={startGame} className="bg-green-500 border-b-8 border-green-700 hover:-translate-y-1 active:border-b-0 active:translate-y-2 text-white font-black text-2xl py-4 px-12 rounded-full shadow-xl flex items-center gap-3 mx-auto transition-all">
                <Play fill="currentColor" size={28}/> COMEÇAR
              </button>
            </div>
          )}

          {gameState === 'dice' && (
            <div className="bg-slate-800/95 backdrop-blur-xl p-8 rounded-[2rem] shadow-2xl border-4 border-slate-600 max-w-sm w-full text-center pointer-events-auto">
              <h3 className="text-white text-2xl font-black uppercase mb-6 drop-shadow-md">Lança o Dado!</h3>
              <DadoUI face={diceValue} isRolling={isRolling} />
              <div className="h-16 mt-6 flex items-center justify-center">
                {!isRolling && !feedback ? (
                  <button onClick={rollDice} className="bg-red-500 border-b-8 border-red-700 hover:bg-red-400 text-white font-black text-xl py-3 px-8 rounded-full w-full active:border-b-0 active:translate-y-2 transition-all">ROLAR DADO</button>
                ) : (
                  <p className="text-green-400 text-2xl font-black animate-pulse">{feedback?.msg || "A rodar..."}</p>
                )}
              </div>
            </div>
          )}

          {gameState === 'event' && specialEvent && (
             <div className="bg-yellow-300/95 backdrop-blur-md p-8 rounded-[2rem] shadow-2xl border-8 border-yellow-500 max-w-lg text-center pointer-events-auto animate-[bounce_1s_ease-in-out_infinite]">
                <AlertTriangle size={80} className={`mx-auto mb-4 ${specialEvent.type === 'reset' ? 'text-red-600' : 'text-orange-600'}`} />
                <h2 className="text-4xl font-black text-slate-800 uppercase mb-4">CASA ESPECIAL!</h2>
                <p className="text-2xl font-bold bg-white p-4 rounded-xl shadow-inner">{specialEvent.text}</p>
             </div>
          )}

          {gameState === 'end' && (
            <div className="bg-yellow-400/95 backdrop-blur-xl p-8 rounded-[2rem] shadow-2xl border-8 border-yellow-600 max-w-xl text-center pointer-events-auto z-50 animate-in zoom-in">
              <div className="text-8xl mb-6 drop-shadow-lg text-white">🏆</div>
              <h2 className="text-5xl font-black text-yellow-900 uppercase mb-4">Vitória!</h2>
              <p className={`text-3xl font-black mb-10 px-6 py-4 rounded-2xl bg-white shadow-inner ${winner === 1 ? 'text-blue-600' : 'text-red-600'}`}>
                O {winner === 1 ? CHARACTERS[p1Char].name.toUpperCase() : CHARACTERS[p2Char].name.toUpperCase()} VENCEU!
              </p>
              <button onClick={() => setGameState('select_p1')} className="bg-slate-800 hover:bg-slate-700 border-b-8 border-slate-900 text-white font-black text-2xl py-4 px-10 rounded-full w-full flex justify-center gap-3 transition-transform hover:-translate-y-1 active:translate-y-2 active:border-b-0">
                <RefreshCw size={28}/> NOVO JOGO
              </button>
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
